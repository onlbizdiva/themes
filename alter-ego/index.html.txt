    <script> 
        // --- INIT ICONS --- 
        lucide.createIcons(); 
 
        // --- PRELOADER --- 
        window.addEventListener('load', () => { 
            const preloader = document.getElementById('preloader'); 
            setTimeout(() => { 
                preloader.classList.add('preloader-done'); 
                setTimeout(() => preloader.remove(), 1000); 
            }, 1200); 
        }); 
 
        // --- CUSTOM CURSOR --- 
        const cursor = document.getElementById('cursor-blend'); 
        let mouseX = window.innerWidth / 2; 
        let mouseY = window.innerHeight / 2; 
        let cX = mouseX; 
        let cY = mouseY; 
 
        document.addEventListener('mousemove', (e) => { 
            mouseX = e.clientX; 
            mouseY = e.clientY; 
        }); 
 
        function animateCursor() { 
            cX += (mouseX - cX) * 0.2; 
            cY += (mouseY - cY) * 0.2; 
            cursor.style.left = cX + 'px'; 
            cursor.style.top = cY + 'px'; 
            requestAnimationFrame(animateCursor); 
        } 
        animateCursor(); 
 
        // Hover States 
        document.querySelectorAll('.hover-target').forEach(el => { 
            el.addEventListener('mouseenter', () => document.body.classList.add('is-hovering')); 
            el.addEventListener('mouseleave', () => document.body.classList.remove('is-hovering')); 
        }); 
 
        // --- SPLIT SCROLL LOGIC --- 
        const colLeft = document.getElementById('col-left'); 
        const colRight = document.getElementById('col-right'); 
        const wrapper = document.querySelector('.scroll-wrapper'); 
        const scrollBadge = document.getElementById('scroll-badge'); 
         
        // Configuration 
        const numSections = 4; 
        // Total transform distance is (numSections - 1) * 100vh = 300vh. 
        // In percentage of the 400vh container, 300vh is 75%. 
        const maxTranslatePct = ((numSections - 1) / numSections) * 100; // 75% 
 
        // Initialize Right Column to start at bottom (showing R1) 
        colRight.style.transform = `translateY(-${maxTranslatePct}%)`; 
 
        let lastScroll = window.scrollY; 
        let rotation = 0; 
 
        function handleScroll() { 
            const scrollY = window.scrollY; 
            const maxScrollHeight = wrapper.offsetHeight - window.innerHeight; 
             
            // Clamp progress between 0 and 1 
            let progress = scrollY / maxScrollHeight; 
            if (progress < 0) progress = 0; 
            if (progress > 1) progress = 1; 
 
            // Left moves UP (0% -> -75%) 
            const leftTranslate = progress * maxTranslatePct; 
            colLeft.style.transform = `translateY(-${leftTranslate}%)`; 
 
            // Right moves DOWN (-75% -> 0%) 
            const rightTranslate = maxTranslatePct - (progress * maxTranslatePct); 
            colRight.style.transform = `translateY(-${rightTranslate}%)`; 
 
            // Badge Rotation Physics based on scroll velocity 
            const velocity = scrollY - lastScroll; 
            rotation += velocity * 0.5; // Spin speed multiplier 
            scrollBadge.style.transform = `translate(-50%, -50%) rotate(${rotation}deg)`; 
 
            lastScroll = scrollY; 
            requestAnimationFrame(() => {}); // Debounce via RAF if needed, but direct assignment 
is usually smooth enough 
        } 
 
        window.addEventListener('scroll', handleScroll, { passive: true }); 
        window.addEventListener('resize', handleScroll); // Recalculate on resize 
         
        // Trigger once to set initial state safely 
        handleScroll(); 
 
    </script>
